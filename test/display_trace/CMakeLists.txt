if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  cmake_minimum_required(VERSION 3.16)
  project(display_trace CXX)
  set(CMAKE_CXX_STANDARD 20)
  enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(DISPLAY_TRACE_RUST "Run the facade through the Rust panel adapter" OFF)
if(DISPLAY_TRACE_RUST)
  set(XP_FFI_HOST_FEATURES "display-panel" CACHE STRING "" FORCE)
  include(${CMAKE_CURRENT_SOURCE_DIR}/../cmake/xp_ffi_host.cmake)
endif()
set(DISPLAY_TRACE_SDK "${CMAKE_CURRENT_SOURCE_DIR}/../../freeink-sdk" CACHE PATH "SDK oracle input")
set(STAGED "${CMAKE_CURRENT_BINARY_DIR}/sdk")
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/prepare.py
  ${DISPLAY_TRACE_SDK} ${STAGED} ${CMAKE_CURRENT_SOURCE_DIR}/stubs RESULT_VARIABLE STAGE_RESULT)
if(NOT STAGE_RESULT EQUAL 0)
  message(FATAL_ERROR "Display oracle SDK staging failed")
endif()
# Regenerate the staging tree whenever an input changes (including mutations).
file(GLOB_RECURSE SDK_INPUTS CONFIGURE_DEPENDS "${DISPLAY_TRACE_SDK}/libs/display/FreeInkDisplay/src/*"
  "${DISPLAY_TRACE_SDK}/libs/display/FreeInkDisplay/include/*")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${SDK_INPUTS}
  "${DISPLAY_TRACE_SDK}/libs/hardware/BoardConfig/include/BoardConfig.h" "${CMAKE_CURRENT_SOURCE_DIR}/prepare.py"
  "${CMAKE_CURRENT_SOURCE_DIR}/stubs/MetalioEInk4Board.h")
foreach(profile IN ITEMS c3 x4pro x4c sticky metalio papermono a4 picco)
  if(profile STREQUAL "c3")
    set(FLAGS FREEINK_DEVICE_X3=1 FREEINK_DEVICE_X4=1)
    set(DRIVERS Ssd1677 Uc8253X3 Uc8279 Uc8179 Uc8279X4)
  elseif(profile STREQUAL "papermono")
    set(FLAGS FREEINK_DEVICE_PAPERMONO=1 BOARD_HAS_PSRAM=1)
    set(DRIVERS PaperMono)
  elseif(profile STREQUAL "a4")
    set(FLAGS FREEINK_DEVICE_EEGO_A4=1 BOARD_HAS_PSRAM=1)
    set(DRIVERS Uc8279cA4)
  else()
    set(DRIVERS Ssd1677)
    if(profile STREQUAL "x4pro")
      set(FLAGS FREEINK_DEVICE_X4PRO=1 BOARD_HAS_PSRAM=1)
      list(APPEND DRIVERS Uc8179 Uc8279X4)
    elseif(profile STREQUAL "x4c")
      set(FLAGS FREEINK_DEVICE_X4CLASSIC=1 BOARD_HAS_PSRAM=1)
      list(APPEND DRIVERS Uc8179 Uc8279X4)
    elseif(profile STREQUAL "metalio")
      set(FLAGS FREEINK_DEVICE_METALIO_EINK4=1 BOARD_HAS_PSRAM=1)
    elseif(profile STREQUAL "sticky")
      set(FLAGS FREEINK_DEVICE_STICKY=1 BOARD_HAS_PSRAM=1)
      list(APPEND DRIVERS Ssd2677)
    elseif(profile STREQUAL "picco")
      set(FLAGS FREEINK_DEVICE_PICCO=1 BOARD_HAS_PSRAM=1)
      list(APPEND DRIVERS Ssd2677)
    endif()
  endif()
  set_source_files_properties(${STAGED}/src/FreeInkDisplay.cpp PROPERTIES COMPILE_OPTIONS "-Wno-unused-variable")
  set_source_files_properties(${STAGED}/src/driver/PaperMonoDriver.cpp PROPERTIES COMPILE_OPTIONS "-Wno-unused-but-set-variable")
  foreach(buffers IN ITEMS single dual)
    set(TARGET display_trace_${profile}_${buffers})
    add_executable(${TARGET} DisplayTrace.cpp Trace.cpp ${STAGED}/src/FreeInkDisplay.cpp ${STAGED}/src/bus/EpdBus.cpp)
    foreach(driver IN LISTS DRIVERS)
      target_sources(${TARGET} PRIVATE ${STAGED}/src/driver/${driver}Driver.cpp)
    endforeach()
    if(DISPLAY_TRACE_RUST)
      target_sources(${TARGET} PRIVATE ${STAGED}/src/driver/RustPanelDriver.cpp)
      target_link_libraries(${TARGET} PRIVATE xp_ffi_host)
      target_compile_definitions(${TARGET} PRIVATE FREEINK_RUST_PANEL=1)
    endif()
    target_include_directories(${TARGET} PRIVATE stubs . ${STAGED} ${STAGED}/include)
    target_compile_definitions(${TARGET} PRIVATE ARDUINO=1 ${FLAGS})
    if(buffers STREQUAL "single")
      target_compile_definitions(${TARGET} PRIVATE EINK_DISPLAY_SINGLE_BUFFER_MODE=1)
    endif()
    if(APPLE)
      target_link_options(${TARGET} PRIVATE "LINKER:-U,_freeink_board_epd_power" "LINKER:-U,_freeink_board_epd_reset")
    endif()
    target_compile_options(${TARGET} PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-unused-function)
  endforeach()
endforeach()
add_test(NAME display_trace COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/oracle.py
  verify --bin-dir ${CMAKE_CURRENT_BINARY_DIR})
add_test(NAME display_trace_sensitivity COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/mutation_check.py
  --work-dir ${CMAKE_CURRENT_BINARY_DIR}/sensitivity)

if(DISPLAY_TRACE_RUST)
  add_executable(display_abi_test DisplayAbiTest.cpp)
  target_link_libraries(display_abi_test PRIVATE xp_ffi_host)
  add_test(NAME display_abi COMMAND display_abi_test)
endif()

# Direct retained-driver fault/state oracle, paired with the C++ build.
foreach(profile IN ITEMS sticky picco papermono)
  set(TARGET retained_trace_${profile})
  if(profile STREQUAL "papermono")
    set(FLAGS FREEINK_DEVICE_PAPERMONO=1 BOARD_HAS_PSRAM=1)
    set(DRIVER PaperMono)
  elseif(profile STREQUAL "sticky")
    set(FLAGS FREEINK_DEVICE_STICKY=1 BOARD_HAS_PSRAM=1)
    set(DRIVER Ssd2677)
  else()
    set(FLAGS FREEINK_DEVICE_PICCO=1 BOARD_HAS_PSRAM=1)
    set(DRIVER Ssd2677)
  endif()
  add_executable(${TARGET} RetainedDriverTrace.cpp Trace.cpp ${STAGED}/src/bus/EpdBus.cpp
    ${STAGED}/src/driver/${DRIVER}Driver.cpp)
  target_include_directories(${TARGET} PRIVATE stubs . ${STAGED} ${STAGED}/include)
  target_compile_definitions(${TARGET} PRIVATE ARDUINO=1 ENABLE_SERIAL_LOG=1 ${FLAGS})
  if(DISPLAY_TRACE_RUST)
    target_sources(${TARGET} PRIVATE ${STAGED}/src/driver/RustPanelDriver.cpp)
    target_link_libraries(${TARGET} PRIVATE xp_ffi_host)
    target_compile_definitions(${TARGET} PRIVATE FREEINK_RUST_PANEL=1)
  endif()
  if(APPLE)
    target_link_options(${TARGET} PRIVATE "LINKER:-U,_freeink_board_epd_power" "LINKER:-U,_freeink_board_epd_reset")
  endif()
endforeach()
if(DISPLAY_TRACE_RUST)
  add_executable(display_retained_abi_test DisplayRetainedAbiTest.cpp)
  target_link_libraries(display_retained_abi_test PRIVATE xp_ffi_host)
  add_test(NAME display_retained_abi COMMAND display_retained_abi_test)
endif()
