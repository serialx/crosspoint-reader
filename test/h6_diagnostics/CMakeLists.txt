cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(h6_diagnostics C CXX)
  set(CMAKE_CXX_STANDARD 20)
  if(NOT REPO_ROOT)
    get_filename_component(REPO_ROOT "../.." ABSOLUTE)
  endif()
  include(${REPO_ROOT}/test/cmake/xp_ffi_host.cmake)
  enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(H6_REFERENCE "" CACHE PATH "Export of 116195b4 for diagnostic comparison")
if(H6_REFERENCE)
  set(SD_REFERENCE ${H6_REFERENCE}/lib/EpdFont)
  set(XTC_REFERENCE ${H6_REFERENCE}/lib/Xtc/Xtc)
else()
  set(SD_REFERENCE ${REPO_ROOT}/test/sdfont_diff/oracle)
  set(XTC_REFERENCE ${REPO_ROOT}/test/xtc_diff/reference)
endif()
foreach(variant IN ITEMS oracle candidate)
  if(variant STREQUAL oracle)
    set(font_dir ${SD_REFERENCE})
  else()
    set(font_dir ${REPO_ROOT}/lib/EpdFont)
  endif()
  add_executable(h6_sd_${variant} ${REPO_ROOT}/test/sdfont_diff/Font.cpp ${font_dir}/SdCardFont.cpp
    ${REPO_ROOT}/lib/EpdFont/EpdFont.cpp ${REPO_ROOT}/lib/Utf8/Utf8.cpp)
  target_include_directories(h6_sd_${variant} PRIVATE stubs ${font_dir}
    ${REPO_ROOT}/test/sd_card_font/stubs ${REPO_ROOT}/lib/EpdFont
    ${REPO_ROOT}/lib/Utf8 ${REPO_ROOT}/lib/Memory)
  target_compile_definitions(h6_sd_${variant} PRIVATE H6_SD_LOGS)
  if(variant STREQUAL candidate)
    target_compile_definitions(h6_sd_${variant} PRIVATE SD_FONT_CANDIDATE)
  endif()
  target_link_libraries(h6_sd_${variant} PRIVATE xp_ffi_host)
endforeach()
set(H6_GENERATED ${CMAKE_CURRENT_BINARY_DIR}/generated)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/instrument.py
  ${REPO_ROOT} ${H6_GENERATED} ${XTC_REFERENCE} COMMAND_ERROR_IS_FATAL ANY)
add_executable(H6Xtc ${H6_GENERATED}/XtcDiagnostic.cpp ${H6_GENERATED}/reference/XtcParser.cpp
  ${REPO_ROOT}/lib/Xtc/Xtc/XtcParser.cpp)
target_include_directories(H6Xtc PRIVATE stubs xtc-stubs ${REPO_ROOT}/test/xtc_diff/stubs ${REPO_ROOT}/lib/Xtc)
target_compile_options(H6Xtc PRIVATE -Wno-logical-op-parentheses)
target_link_libraries(H6Xtc PRIVATE xp_ffi_host)
add_test(NAME H6XtcGenerate COMMAND ${Python3_EXECUTABLE} ${REPO_ROOT}/test/xtc_diff/generate.py ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
set_tests_properties(H6XtcGenerate PROPERTIES FIXTURES_SETUP h6_xtc_files)
add_test(NAME H6XtcDiagnostics COMMAND H6Xtc ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
set_tests_properties(H6XtcDiagnostics PROPERTIES FIXTURES_REQUIRED h6_xtc_files)
add_test(NAME H6SdDiagnostics COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/sd.py
  ${REPO_ROOT} $<TARGET_FILE_DIR:h6_sd_candidate>)

if(NOT TARGET TextGolden)
  add_subdirectory(${REPO_ROOT}/test/text_golden ${CMAKE_CURRENT_BINARY_DIR}/text-golden)
endif()
get_target_property(H6_GFX_SOURCES TextGolden SOURCES)
list(FILTER H6_GFX_SOURCES EXCLUDE REGEX "TextGolden.cpp$")
foreach(source IN LISTS H6_GFX_SOURCES)
  if(source MATCHES "/instrumented/")
    set_source_files_properties(${source} PROPERTIES GENERATED TRUE)
  endif()
endforeach()
add_executable(H6Gfx Gfx.cpp ${H6_GFX_SOURCES})
add_dependencies(H6Gfx TextGolden)
get_target_property(H6_GFX_INCLUDES TextGolden INCLUDE_DIRECTORIES)
get_target_property(H6_GFX_DEFINES TextGolden COMPILE_DEFINITIONS)
get_target_property(H6_GFX_OPTIONS TextGolden COMPILE_OPTIONS)
get_target_property(H6_GFX_LINK_OPTIONS TextGolden LINK_OPTIONS)
target_include_directories(H6Gfx PRIVATE stubs ${H6_GFX_INCLUDES} ${REPO_ROOT}/test/text_golden/stubs ${REPO_ROOT}/test/text_golden)
target_compile_definitions(H6Gfx PRIVATE ${H6_GFX_DEFINES})
target_compile_options(H6Gfx PRIVATE ${H6_GFX_OPTIONS})
target_link_options(H6Gfx PRIVATE ${H6_GFX_LINK_OPTIONS})
target_link_libraries(H6Gfx PRIVATE xp_ffi_host)
add_test(NAME H6GfxDiagnostics COMMAND H6Gfx)
