cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(line_render_golden C CXX)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_BUILD_TYPE Release)
  get_filename_component(REPO_ROOT "../.." ABSOLUTE)
  enable_testing()
  include(../cmake/xp_ffi_host.cmake)
  add_subdirectory(../page_golden page_golden)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_target_property(PAGE_SOURCES PageGolden SOURCES)
get_target_property(PAGE_INCLUDES PageGolden INCLUDE_DIRECTORIES)
get_target_property(PAGE_DEFINES PageGolden COMPILE_DEFINITIONS)
get_target_property(PAGE_OPTIONS PageGolden COMPILE_OPTIONS)
get_target_property(PAGE_LINK_OPTIONS PageGolden LINK_OPTIONS)
list(FILTER PAGE_SOURCES EXCLUDE REGEX "PageGolden.cpp$")
set(LINE_SOURCES)
foreach(SOURCE IN LISTS PAGE_SOURCES)
  if(NOT IS_ABSOLUTE "${SOURCE}")
    set(SOURCE "${REPO_ROOT}/test/page_golden/${SOURCE}")
  endif()
  if(SOURCE MATCHES "/instrumented/")
    set_source_files_properties(${SOURCE} PROPERTIES GENERATED TRUE)
  endif()
  list(APPEND LINE_SOURCES "${SOURCE}")
endforeach()
set(EXTRACTED ${CMAKE_CURRENT_BINARY_DIR}/Consumers.inc)
add_custom_command(OUTPUT ${EXTRACTED}
  COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/extract.py ${REPO_ROOT} ${EXTRACTED}
  DEPENDS extract.py ${REPO_ROOT}/src/activities/reader/DictionaryDefinitionActivity.cpp
    ${REPO_ROOT}/src/activities/reader/ClipSelectionActivity.cpp
    ${REPO_ROOT}/src/activities/reader/ClipSelectionActivity.h VERBATIM)
add_executable(LineRenderGolden LineRenderGolden.cpp ${LINE_SOURCES} ${EXTRACTED}
  ${REPO_ROOT}/src/activities/settings/TextSettingsPreview.cpp)
target_include_directories(LineRenderGolden BEFORE PRIVATE stubs ${CMAKE_CURRENT_BINARY_DIR})
target_include_directories(LineRenderGolden PRIVATE ${PAGE_INCLUDES} ${REPO_ROOT}/test/page_golden)
target_compile_definitions(LineRenderGolden PRIVATE ${PAGE_DEFINES})
target_compile_options(LineRenderGolden PRIVATE ${PAGE_OPTIONS})
target_link_options(LineRenderGolden PRIVATE ${PAGE_LINK_OPTIONS})
target_link_libraries(LineRenderGolden PRIVATE xp_ffi_host)
add_dependencies(LineRenderGolden PageGolden)
add_test(NAME LineRenderGolden COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/run.py check
  --binary $<TARGET_FILE:LineRenderGolden> --work-dir ${CMAKE_CURRENT_BINARY_DIR}/check)
