if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  cmake_minimum_required(VERSION 3.20)
  project(LogLiteTests LANGUAGES CXX)
  enable_testing()
  set(CMAKE_CXX_STANDARD 20)
  set(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
endif()
add_executable(LogLiteTest LogLiteTest.cpp ${REPO_ROOT}/lib/Logging/BoundedLog.cpp)
target_include_directories(LogLiteTest PRIVATE ${REPO_ROOT}/lib/Logging)
target_compile_options(LogLiteTest PRIVATE -Wall -Wextra -Werror -pedantic)
add_test(NAME log_lite COMMAND LogLiteTest)
foreach(cores IN ITEMS 1 2)
  add_executable(LogCriticalSectionTest${cores} LogCriticalSectionTest.cpp)
  target_include_directories(LogCriticalSectionTest${cores} PRIVATE stubs ${REPO_ROOT}/lib/Logging)
  target_compile_definitions(LogCriticalSectionTest${cores} PRIVATE configNUM_CORES=${cores})
  target_compile_options(LogCriticalSectionTest${cores} PRIVATE -Wall -Wextra -Werror -pedantic)
  add_test(NAME log_lite_critical_${cores} COMMAND LogCriticalSectionTest${cores})
endforeach()
add_executable(LoggingIntegrationTest LoggingIntegrationTest.cpp
  ${REPO_ROOT}/lib/Logging/Logging.cpp ${REPO_ROOT}/lib/Logging/BoundedLog.cpp)
target_include_directories(LoggingIntegrationTest PRIVATE stubs ${REPO_ROOT}/lib/Logging)
target_compile_definitions(LoggingIntegrationTest PRIVATE configNUM_CORES=1 ENABLE_SERIAL_LOG)
target_compile_options(LoggingIntegrationTest PRIVATE -Wall -Wextra -Werror -pedantic)
add_test(NAME log_lite_integration COMMAND LoggingIntegrationTest)
