cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(xtc_diff C CXX)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type")
  set(CMAKE_CXX_STANDARD 20)
  set(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
  include(../cmake/xp_ffi_host.cmake)
  enable_testing()
endif()
add_executable(XtcDiff XtcDiff.cpp reference/XtcParser.cpp ${REPO_ROOT}/lib/Xtc/Xtc/XtcParser.cpp)
target_include_directories(XtcDiff PRIVATE stubs ${REPO_ROOT}/lib/Xtc)
target_link_libraries(XtcDiff PRIVATE xp_ffi_host)
target_compile_options(XtcDiff PRIVATE -Wall -Wextra -Wno-logical-op-parentheses)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME xtc_generate COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/generate.py ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
set_tests_properties(xtc_generate PROPERTIES FIXTURES_SETUP xtc_files)
add_test(NAME xtc_diff COMMAND XtcDiff ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
set_tests_properties(xtc_diff PROPERTIES FIXTURES_REQUIRED xtc_files)
