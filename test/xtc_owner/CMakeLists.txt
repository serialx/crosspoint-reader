cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(xtc_owner LANGUAGES C CXX)
  set(CMAKE_CXX_STANDARD 20)
  get_filename_component(REPO_ROOT "../.." ABSOLUTE)
  enable_testing()
  include(${REPO_ROOT}/test/cmake/xp_ffi_host.cmake)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_executable(XtcOwnerContract Contract.cpp
  ${REPO_ROOT}/lib/Xtc/Xtc/XtcParser.cpp ${REPO_ROOT}/lib/Xtc/Xtc/RustXtcParser.cpp)
target_include_directories(XtcOwnerContract PRIVATE stubs ${REPO_ROOT}/test/book_io/stubs ${REPO_ROOT}/lib/Xtc)
target_link_libraries(XtcOwnerContract PRIVATE xp_ffi_host)
add_test(NAME xtc_owner_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/run.py $<TARGET_FILE:XtcOwnerContract>)
add_test(NAME xtc_owner_rust COMMAND cargo test --manifest-path ${CMAKE_CURRENT_SOURCE_DIR}/rust/Cargo.toml --offline --locked)
add_test(NAME xtc_owner_oracle_negative COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/oracle_negative.py $<TARGET_FILE:XtcOwnerContract>)
add_executable(XtcOwnerAbi Abi.c)
target_include_directories(XtcOwnerAbi PRIVATE ${REPO_ROOT}/rust/crates/xp-ffi/include)
add_test(NAME xtc_owner_c_abi COMMAND XtcOwnerAbi)
add_test(NAME xtc_owner_oracle_integrity COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/oracle_integrity.py)
add_executable(XtcOwnerSwitch Switch.cpp)
target_include_directories(XtcOwnerSwitch PRIVATE ${REPO_ROOT}/test/book_io/stubs ${REPO_ROOT}/lib/Xtc ${REPO_ROOT}/rust/crates/xp-ffi/include)
add_test(NAME xtc_owner_switch COMMAND XtcOwnerSwitch)
add_test(NAME xtc_owner_boundaries COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/boundary.py --out ${CMAKE_CURRENT_BINARY_DIR}/boundaries)

add_test(NAME HalContract.xtc_rust_slots COMMAND ${XP_FFI_CARGO} test
  --manifest-path ${CMAKE_CURRENT_SOURCE_DIR}/rust/Cargo.toml --offline --locked --bin xtc-boundary)
